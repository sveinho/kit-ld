TRANSFORMASJON: Fra flat @graph til nøstet tre-struktur (Før visning i editor) function transformToEditorFormat(jsonLd) { currentContext = jsonLd["@context"] || {}; const graph = jsonLd["@graph"] || []; const catalog = graph.find(item => item.type === "dcat:Catalog"); if (!catalog) { alert("Kunne ikke finne et dcat:Catalog-element i filen."); return jsonLd; } const dataServices = graph.filter(item => item.type === "dcat:DataService"); const datasets = graph.filter(item => item.type === "dcat:Dataset"); const distributions = graph.filter(item => item.type === "dcat:Distribution"); const agents = graph.filter(item => item.type === "foaf:Agent"); if (typeof catalog.publisher === "string") { const foundAgent = agents.find(a => a.id === catalog.publisher); if (foundAgent) catalog.publisher = foundAgent; } if (Array.isArray(catalog.service)) { catalog.service = catalog.service.map(serviceUri => { const foundService = dataServices.find(s => s.id === serviceUri); if (foundService) { const cleanService = {...foundService}; delete cleanService.servesDataset; return cleanService; } return serviceUri; }); } if (Array.isArray(catalog.dataset)) { catalog.dataset = catalog.dataset.map(datasetUri => { const foundDataset = datasets.find(d => d.id === datasetUri); if (foundDataset) { const nestedDataset = {...foundDataset}; if (Array.isArray(nestedDataset.distribution)) { nestedDataset.distribution = nestedDataset.distribution.map(distUri => { const foundDist = distributions.find(dist => dist.id === distUri); return foundDist || distUri; }); } return nestedDataset; } return datasetUri; }); } return { "catalog": catalog }; } // 2. TRANSFORMASJON: Fra nøstet tre-struktur tilbake til flat @graph (Før eksport) function transformToValidJsonLd(editorJson) { if (!editorJson.catalog) return editorJson; const catalog = JSON.parse(JSON.stringify(editorJson.catalog)); const graph = []; if (catalog.publisher && typeof catalog.publisher === "object") { graph.push(catalog.publisher); catalog.publisher = catalog.publisher.id; } if (Array.isArray(catalog.service)) { catalog.service = catalog.service.map(service => { if (typeof service === "object" && service.id) { service.servesDataset = [catalog.id]; graph.push(service); return service.id; } return service; }); } if (Array.isArray(catalog.dataset)) { catalog.dataset = catalog.dataset.map(dataset => { if (typeof dataset === "object" && dataset.id) { if (Array.isArray(dataset.distribution)) { dataset.distribution = dataset.distribution.map(dist => { if (typeof dist === "object" && dist.id) { graph.push(dist); return dist.id; } return dist; }); } graph.push(dataset); return dataset.id; } return dataset; }); } graph.push(catalog); return { "@context": currentContext, "@graph": graph }; } // Lytte etter filopplasting document.getElementById('file-upload').addEventListener('change', function(event) { const file = event.target.files[0];
